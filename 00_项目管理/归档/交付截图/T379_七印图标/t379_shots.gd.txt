var out_dir := ""


func _ready() -> void:
	for arg in OS.get_cmdline_user_args():
		if arg.begins_with("--out="):
			out_dir = arg.substr(6)
	_run.call_deferred()


func _frames(n := 10) -> void:
	for i in range(n):
		await get_tree().process_frame


func _shot(name: String) -> void:
	await _frames(3)
	await RenderingServer.frame_post_draw
	DirAccess.make_dir_recursive_absolute(out_dir)
	get_viewport().get_texture().get_image().save_png(out_dir.path_join(name + ".png"))
	print("SHOT ", name)


func _altar(seal_idx: int) -> XTianqiSkinAltar:
	var lineup: Array = XTier.ids(XTier.WEAK).filter(func(id): return not XImportedBalance.is_disabled_actor(id) and not XActorModules.IDS.has(id)).slice(0, 5)
	var s := XTianqi.start_new_game(3, "jianxian", lineup)
	XTianqi.take_pending()
	for id in lineup:
		XTianqi._joining = true
		s.add_actor(int(id))
		XTianqi._joining = false
	s.map_id = 4
	s.set_variable(20, 36)
	XState.current = s
	s.flags.tianqi["line"] = "war"
	XTianqiWarLine.progress(s)["stage"] = XTianqiWarLine.STAGE_DONE
	XTianqiQiyin.set_current("")
	XTianqiSkinAltar.last_seal = ""
	XTianqiQiyin.sync(s)
	var a := XTianqiSkinAltar.new()
	a.setup(s, XTianqiQiyin.altar_options(s))
	a.set_anchors_and_offsets_preset(Control.PRESET_FULL_RECT)
	add_child(a)
	await _frames(20)
	a.select(seal_idx)
	a.toggle_rules()
	await _frames(10)
	return a


func _run() -> void:
	if out_dir == "" or DisplayServer.get_name() == "headless":
		get_tree().quit()
		return
	XSplashScreen.shown = true
	XConfig.ensure_loaded()
	XConfig._values["codexUiSkin"] = true
	XUiSkin.ENABLED = true
	XUiSkin.FORCE = true
	XUiSkin.motion_override = XUiSkin.Motion.OFF
	var win := get_window()
	for wide in [false, true]:
		win.size = Vector2i(1000, 450) if wide else Vector2i(800, 450)
		if wide:
			XSkinLayout.window_override = Vector2(1000, 450)
		await _frames(10)
		for i in [6, 1]:
			var a := await _altar(i)
			if wide:
				XSkinLayout._apply_wide(get_tree().root)
				await _frames(10)
			await _shot("altar_rule_%s_%d" % ["20x9" if wide else "16x9", i])
			a.queue_free()
			await _frames(3)
	get_tree().quit()
